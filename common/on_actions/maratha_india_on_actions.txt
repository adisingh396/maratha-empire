on_actions = {
	on_startup = {
		effect = {
			# Ensure Indian tags have 50 compliance integration like Kaiserreich
			every_state = {
				limit = {
					OR = {
						is_core_of = MAR
						is_core_of = HYD
						is_core_of = MYS
						is_core_of = RJP
					}
				}
				enable_state_integration_50_compliance = yes
			}
		}
	}
	on_annex = {
		effect = {
			if = {
				limit = {
					OR = {
						tag = MAR
						tag = HYD
						tag = RJP
						tag = MYS
						FROM = { OR = { tag = MAR tag = HYD tag = RJP } }
					}
					any_owned_state = { is_core_of = MAR }
				}
				# Transfer characters like Kaiserreich HYD/MYS
				FROM = {
					every_character = {
						limit = { has_character_flag = HYD_character }
						set_nationality = HYD
					}
					every_character = {
						limit = { has_character_flag = MYS_character }
						set_nationality = MYS
					}
				}
				# Fire unification event after 1 day
				hidden_effect = {
					country_event = { id = MAR_news.2 days = 1 }
				}
			}
			# Revolution check: if stability <0.3, fire revolt
			if = {
				limit = {
					FROM = { has_stability < 0.30 }
					NOT = { tag = MAR }
				}
				FROM = { country_event = { id = MAR_revolt.1 days = 7 } }
			}
		}
	}
	on_daily = {
		effect = {
			# Rare MAR revolution if not strong (stability <0.2 and war_support <0.2)
			if = {
				limit = {
					tag = MAR
					has_stability < 0.20
					has_war_support < 0.20
					has_game_rule = { rule = MAR_revolts value = yes }
				}
				country_event = { id = MAR_revolt.10 days = 30 random_days = 15 }
			}
		}
	}
}
